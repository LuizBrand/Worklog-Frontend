```ts
const CNPJ_D1_WEIGHTS = [5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2]
const CNPJ_D2_WEIGHTS = [6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2]

export function isValidCnpj(value: string): boolean {
  const cnpj = stripDocumento(value)
  if (cnpj.length !== 14) return false
  if (!/^[0-9A-Z]{12}\d{2}$/.test(cnpj)) return false   // DV é sempre dígito
  if (/^(.)\1{13}$/.test(cnpj)) return false            // repetido é inválido

  const values = cnpj.split('').map((c) => c.charCodeAt(0) - 48)
  const checkDigit = (weights: number[]) => {
    let sum = 0
    for (let i = 0; i < weights.length; i++) sum += values[i] * weights[i]
    const rest = sum % 11
    return rest < 2 ? 0 : 11 - rest
  }
  if (checkDigit(CNPJ_D1_WEIGHTS) !== values[12]) return false
  return checkDigit(CNPJ_D2_WEIGHTS) === values[13]
}
```

```ts
export function apiFieldPathToRhf(key: string): string {

export function applyApiFieldErrors<T extends FieldValues>(
  err: unknown,
  setError: UseFormSetError<T>,
): boolean {
  const fieldErrors = getApiErrorBody(err)?.fieldErrors
  if (!fieldErrors) return false
  let applied = false
  for (const [key, message] of Object.entries(fieldErrors)) {
    if (!message) continue
    setError(apiFieldPathToRhf(key) as FieldPath<T>, { type: 'server', message: String(message) })
    applied = true
  }
  return applied
}
```

### Gotchas descobertos nesta sessão (candidatos a `memory/gotchas.md`)

- **`page.addInitScript` roda em TODA navegação.** Usar para setar o tema e depois trocar
  via `evaluate` + `reload` faz o init script reescrever o valor antigo — a captura "light"
  sai dark, silenciosamente. Setar tema com `evaluate` + `reload`, sem init script.
- **Asserção de tema tem que comparar o valor computado.** Light é `:root`
  (`--wl-bg: #f2f2f4`), dark é a classe `.dark` (`#0b0b10`). Um check `!!bg` passa mesmo com
  o tema errado.
- **`browser.newPage()` cria contexto novo e perde o cookie de sessão.** Para uma segunda
  aba autenticada, usar `page.context().newPage()` ou reusar a mesma page.
- **`page.goto(..., { waitUntil: 'domcontentloaded' })` no login** clica antes da hidratação
  e o submit não acontece. Usar `networkidle`.
- **`page.waitForURL` espera o evento `load`**, que não refira em redirect client-side do
  App Router. Usar `waitForFunction` em `location.pathname`.
- **MCP do Playwright não funciona neste ambiente** — canal `chrome` ausente. Usar o
  `playwright` das devDependencies, **rodando da raiz do projeto**.
- **`useLookupByCnpj` é `useQuery`**, não mutation — lookup manual precisa
  `enabled: false` + `refetch()`.
- **Prefix-match de query key do React Query é por elemento do array.**
  `['/clients/abc']` não invalida `['/clients/abc/branches']`.

### Armadilhas específicas deste módulo

- **`PATCH /clients` ignora `branches`.** Toda manutenção de filial, incluindo a matriz, é
  pelo sub-recurso. Editar a matriz junto do cliente exige 2 requests.
- **`contatos` no PATCH de filial SUBSTITUI a lista inteira** e recria os contatos, então
  os `publicId` deles mudam. Mandar sempre a lista completa. Não guardar referência de
  contato entre operações.
- **`isMatriz: true` no POST de filial volta 422.** Promover é operação do PATCH.
- **A matriz nunca pode ser inativada nem rebaixada.** Promover outra primeiro.
- **`documento` é único GLOBALMENTE**, inclusive de filial inativa. Inativar não libera o
  CNPJ — vale texto explicativo no diálogo de confirmação.
- **Assimetria 400/422:** checksum inválido volta **400 com `fieldErrors`** no
  `POST /clients/` e **422 com mensagem solta** no `POST /branches`. Tratar os dois.
- **Mensagens de 404/409 são em inglês e contêm IDs** — não exibir cru. As de 422 são
  pt-BR e podem ir direto para a tela.
- **Nunca adicionar header customizado** — o CORS bloqueia no preflight.
- **`var(--wl-surface-1)` não existe.** São `--wl-surface`, `--wl-surface-2`, `--wl-bg`.
  Conferir `src/app/globals.css`.
- **Hook de evidência visual exige os 5 campos com dois-pontos na mesma linha**
  (`**Changed files:** ...`). Título markdown sem `:` faz o Stop hook rejeitar sem
  explicar o motivo.
